CSTAR = -0.999786375236
THRESH = 0.026701304157334
EPS_N = 1e-6
EPS_W = 1e-6

CLASS_ORDER = [

STRUCTURED_SOURCE_MODES = {'structured_permuted_collapse','structured_anti_collapse','novel_signed_scramble_collapse','novel_source_phase_flip'}
REJECTED_SOURCE_MODES = {'wrong_sign','random','hold_broken','novel_noise_sink'}


def smooth(x, sigma):
    if gaussian_filter is not None:
        return gaussian_filter(x, sigma=sigma, mode='wrap')
    y = x.copy()
    for _ in range(max(1, int(round(sigma * 3)))):
        y = (y + np.roll(y, 1, 0) + np.roll(y, -1, 0) + np.roll(y, 1, 1) + np.roll(y, -1, 1)) / 5.0
    return y


def normalize(x):
    return (x - float(np.mean(x))) / (float(np.std(x)) + 1e-12)


def grad(f):
    return (np.roll(f, -1, 0) - np.roll(f, 1, 0)) / 2.0, (np.roll(f, -1, 1) - np.roll(f, 1, 1)) / 2.0


def div(vx, vy):
    return (np.roll(vx, -1, 0) - np.roll(vx, 1, 0)) / 2.0 + (np.roll(vy, -1, 1) - np.roll(vy, 1, 1)) / 2.0


def soft_A(psi):
    z = psi - float(np.max(psi))
    A = np.exp(z)
    return A / (float(np.mean(A)) + 1e-12)


def metrics_field(psi, phi):
    px, py = grad(psi); fx, fy = grad(phi)
    ap = np.sqrt(px * px + py * py + EPS_N * EPS_N)
    af = np.sqrt(fx * fx + fy * fy + EPS_N * EPS_N)
    npx, npy = px / ap, py / ap
    nfx, nfy = fx / af, fy / af
    q = npx * nfx + npy * nfy
    A = soft_A(psi)
    a = A * ap + EPS_W
    w = a / np.sum(a)
    m = float(np.sum(w * q))
    var = float(np.sum(w * (q - m) ** 2))
    dac = float(np.sqrt(max(var, 0.0)))
    eosc = float(0.5 * dac * dac + 0.5 * (m - CSTAR) ** 2)
    return dict(E_OSC=eosc, D_AC=dac, m_AC=m, orientation_error=abs(m - CSTAR),
                A_cv=float(np.std(A) / (np.mean(A) + 1e-12)), max_weight_frac=float(np.max(w)),
                basin=bool(eosc <= THRESH))


def energy(psi, phi):
    return metrics_field(psi, phi)['E_OSC']


def semi_derivative(psi, phi):
    px, py = grad(psi); fx, fy = grad(phi)
    ap = np.sqrt(px * px + py * py + EPS_N * EPS_N)
    af = np.sqrt(fx * fx + fy * fy + EPS_N * EPS_N)
    npx, npy = px / ap, py / ap
    nfx, nfy = fx / af, fy / af
    q = npx * nfx + npy * nfy
    A = soft_A(psi)
    w = A * ap + EPS_W
    w = w / np.sum(w)
    coeff = w * (q - CSTAR)
    vx = coeff * (nfx - q * npx) / ap
    vy = coeff * (nfy - q * npy) / ap
    gpsi = -div(vx, vy)
    ux = coeff * (npx - q * nfx) / af
    uy = coeff * (npy - q * nfy) / af
    gphi = -div(ux, uy)
    return gpsi, gphi


def make_defect(seed, n, geometry):
    rng = np.random.default_rng(seed)
    x = np.linspace(-1, 1, n, endpoint=False)
    X, Y = np.meshgrid(x, x, indexing='ij')
    if geometry == 'oblique_triple':
        mu = np.exp(-((X-.55)**2+(Y+.10)**2)/0.035) - 0.8*np.exp(-((X+.36)**2+(Y-.35)**2)/0.055) + .45*np.exp(-((X-.05)**2+(Y-.55)**2)/0.07)
    elif geometry == 'ripple_wall':
        mu = np.exp(-(X + .55 + .12*np.sin(4*np.pi*Y))**2/0.020) - .35*np.exp(-(Y-.30)**2/0.06)
    elif geometry == 'broken_ring':
        R = np.sqrt((X-.08)**2 + (Y+.02)**2)
        gap = 1.0 - 0.55*np.exp(-((X+.48)**2+(Y-.05)**2)/0.06)
        mu = gap*np.exp(-(R-.50)**2/0.012) - .38*np.exp(-(R-.22)**2/0.018)
    elif geometry == 'corner_sink':
        mu = np.exp(-((X+.78)**2+(Y+.72)**2)/0.025) - .65*np.exp(-((X-.15)**2+(Y-.20)**2)/0.050)
    elif geometry == 'prealigned_low_energy':
        mu = np.exp(-((X-.2)**2+(Y+.25)**2)/0.045) - .50*np.exp(-((X+.3)**2+(Y-.35)**2)/0.065)
    elif geometry == 'phase_fault':
        mu = np.sin(3*np.pi*X + 1.7*np.pi*Y) * np.exp(-0.45*(X*X+Y*Y)) + .35*np.exp(-((X-.58)**2+(Y+.45)**2)/0.035)
    else:
        mu = rng.normal(size=(n, n))
    return normalize(smooth(mu, max(.45, n / 58)))


def make_initial(seed, n, geometry, mode):
    rng = np.random.default_rng(seed + 1777 * n)
    defect = make_defect(seed, n, geometry)
    base = smooth(rng.normal(size=(n, n)), max(.8, n / 11.5))
    psi = normalize(base - 0.48 * defect + 0.12 * smooth(rng.normal(size=(n, n)), max(.6, n / 30)))
    if geometry == 'prealigned_low_energy' and mode in ACTIVE_SOURCE_MODES | {'hold_broken'} | PASSIVE_SOURCE_MODES:
        phi = normalize(-psi + 0.003 * smooth(rng.normal(size=(n, n)), max(.8, n / 17)))
    elif mode in PASSIVE_SOURCE_MODES:
        phi = normalize(-psi + 0.004 * smooth(rng.normal(size=(n, n)), max(.8, n / 16)))
    else:
        phi = normalize(smooth(rng.normal(size=(n, n)), max(.8, n / 10.2)) + 0.22 * defect)
    return psi, phi, defect


def permute_field(x, seed):
    rng = np.random.default_rng(seed)
    flat = x.ravel().copy(); rng.shuffle(flat)
    return flat.reshape(x.shape)


def unit_pair(dpsi, dphi):
    nn = float(np.sqrt(np.mean(dpsi * dpsi) + np.mean(dphi * dphi))) + 1e-12
    return dpsi / nn, dphi / nn


def run_case(seed, n, geometry, mode, max_steps=64):
    psi, phi, defect = make_initial(seed, n, geometry, mode)
    rows = []
    accepted = 0
    for step in range(max_steps + 1):
        met = metrics_field(psi, phi)
        rows.append(dict(seed=seed, n=n, geometry=geometry, mode=mode, step=step, accepted_steps=accepted, **met))
        if mode in ACTIVE_SOURCE_MODES and met['E_OSC'] <= THRESH:
            break
        if mode in PASSIVE_SOURCE_MODES:
            break
        if step == max_steps:
            break
        gpsi, gphi = semi_derivative(psi, phi)
        gn = float(np.sqrt(np.mean(gpsi*gpsi) + np.mean(gphi*gphi))) + 1e-12
        require_descent = False
        base_alpha = 0.5
        if mode == 'directed_variational':
            dpsi, dphi = -gpsi / gn, -gphi / gn; require_descent = True; base_alpha = 1.0
        elif mode == 'defect_forced':
            dpsi, dphi = unit_pair(-0.05 * defect, defect); base_alpha = 1.2
        elif mode == 'novel_local_defect_repair':
            target_phi = -psi + 0.02 * smooth(defect, max(.4, n/70))
            dpsi, dphi = unit_pair(-0.04 * smooth(defect, .8), target_phi - phi); base_alpha = 0.9
        elif mode == 'novel_gradient_shear':
            dx, dy = grad(defect)
            dpsi, dphi = unit_pair(-0.03 * dx + 0.02 * dy, (-psi - phi) + 0.04 * dx); base_alpha = 0.75
        elif mode == 'structured_permuted_collapse':
            pdef = permute_field(defect, seed + 23003)
            target_phi = -psi + 0.10 * pdef
            dpsi, dphi = unit_pair(0.015 * pdef, target_phi - phi); base_alpha = 0.18
        elif mode == 'structured_anti_collapse':
            target_phi = -psi - 0.12 * defect
            dpsi, dphi = unit_pair(0.015 * defect, target_phi - phi); base_alpha = 0.18
        elif mode == 'novel_source_phase_flip':
            # structured source that usually collapses endpoint but with nonlocal source phase.
            rot = np.roll(defect, shift=n//3, axis=0) - np.roll(defect, shift=n//4, axis=1)
            target_phi = -psi + 0.08 * rot
            dpsi, dphi = unit_pair(0.014 * rot, target_phi - phi); base_alpha = 0.16
        elif mode == 'novel_signed_scramble_collapse':
            # structured source family; in phase_fault geometry this deliberately fails basin entry.
            scr = permute_field(np.sign(defect) * np.sqrt(np.abs(defect)+1e-9), seed + 777)
            if geometry == 'phase_fault':
                target_phi = phi + 0.15 * scr   # fails to align with -psi
                base_alpha = 0.10
            else:
                target_phi = -psi + 0.12 * scr
                base_alpha = 0.15
            dpsi, dphi = unit_pair(0.012 * scr, target_phi - phi)
        elif mode == 'wrong_sign':
            dpsi, dphi = gpsi / gn, gphi / gn; base_alpha = 0.8
        elif mode == 'random':
            rng = np.random.default_rng(seed * 1009 + step * 47 + n)
            dpsi = smooth(rng.normal(size=(n, n)), max(.55, n / 48))
            dphi = smooth(rng.normal(size=(n, n)), max(.55, n / 48))
            dpsi, dphi = unit_pair(dpsi, dphi); base_alpha = 0.45
        elif mode == 'novel_noise_sink':
            rng = np.random.default_rng(seed * 881 + step * 19 + n)
            sink = smooth(rng.normal(size=(n,n)), max(.45, n/60))
            dpsi, dphi = unit_pair(sink, -sink); base_alpha = 0.35
        elif mode == 'hold_broken':
            continue
        else:
            raise ValueError(mode)
        e0 = met['E_OSC']
        alpha = base_alpha
        accepted_this = False
        for _ in range(18):
            p2 = normalize(psi + alpha * dpsi)
            f2 = normalize(phi + alpha * dphi)
            e2 = energy(p2, f2)
            if (not require_descent) or e2 <= e0 + 1e-12:
                psi, phi = p2, f2
                accepted += 1
                accepted_this = True
                break
            alpha *= 0.5
        if require_descent and not accepted_this:
            break
    return rows


def source_family(mode: str) -> str:
    if mode in ACTIVE_SOURCE_MODES: return 'active_source'
    if mode in PASSIVE_SOURCE_MODES: return 'passive_source'
    if mode in STRUCTURED_SOURCE_MODES: return 'structured_source'
    return 'rejected_or_broken_source'


def seven_class(row: pd.Series, fam_col='source_family') -> str:
    fam = row[fam_col]
    mode = row['mode']
    start = bool(row['start_E_basin'])
    endpoint = bool(row['endpoint_E_basin'])
    acc = int(row['accepted_steps'])
    red = float(row['E_reduction_frac'])
    if fam == 'active_source':
        if start and endpoint and acc == 0:
            return 'active_initial_basin_no_repair_needed'
        if endpoint and acc > 0 and red > 0:
            return 'active_repair_source_compatible'
        return 'outside_basin_rejected'
    if fam == 'passive_source':
        if mode in PASSIVE_SOURCE_MODES and start and endpoint and acc == 0 and abs(red) < 1e-12:
            return 'passive_no_motion_basin_member'
        return 'outside_basin_rejected'
    if fam == 'structured_source':
        if endpoint:
            return 'structured_endpoint_collapse_not_legit'
        return 'structured_source_failed_basin_entry'
    if start and endpoint and acc == 0:
        return 'initial_basin_broken_hold_not_passive'
    return 'outside_basin_rejected'


def binary_scores(y_true, y_pred):
    y_true = np.asarray(y_true, dtype=bool); y_pred = np.asarray(y_pred, dtype=bool)
    tp = int(np.sum(y_true & y_pred)); fp = int(np.sum(~y_true & y_pred)); tn = int(np.sum(~y_true & ~y_pred)); fn = int(np.sum(y_true & ~y_pred))
    rec = tp/(tp+fn) if tp+fn else 0.0
    spec = tn/(tn+fp) if tn+fp else 0.0
    return dict(TP=tp, FP=fp, TN=tn, FN=fn, precision=tp/(tp+fp) if tp+fp else 0.0, recall=rec, specificity=spec, accuracy=(tp+tn)/(tp+fp+tn+fn) if tp+fp+tn+fn else 0.0, balanced_accuracy=0.5*(rec+spec))


def confusion(true, pred):
    mat = pd.DataFrame(0, index=CLASS_ORDER, columns=CLASS_ORDER, dtype=int)
    for t, p in zip(true, pred):
        mat.loc[t, p] += 1
    return mat


def per_class_metrics(true, pred):
    rows=[]
    true=np.asarray(true,dtype=object); pred=np.asarray(pred,dtype=object)
    for lab in CLASS_ORDER:
        r=binary_scores(true==lab, pred==lab)
        r['class']=lab; r['support']=int(np.sum(true==lab))
        rows.append(r)
    return pd.DataFrame(rows)[['class','support','TP','FP','TN','FN','precision','recall','specificity','accuracy','balanced_accuracy']]


def main():

    ep = tr.sort_values('step').groupby(['seed','n','geometry','mode'], as_index=False).tail(1).copy()
    ep = ep.merge(starts,on=['seed','n','geometry','mode'],how='left')
    ep['E_reduction_frac']=(ep.start_E_OSC-ep.E_OSC)/(ep.start_E_OSC+1e-12)
    ep['D_reduction_frac']=(ep.start_D_AC-ep.D_AC)/(ep.start_D_AC+1e-12)
    ep['endpoint_E_basin']=ep.E_OSC<=THRESH
    ep['start_E_basin']=ep.start_E_OSC<=THRESH

    for key,g in tr.groupby(['seed','n','geometry','mode']):
        gg=g.sort_values('step')
        e=gg.E_OSC.to_numpy(); d=gg.D_AC.to_numpy(); o=gg.orientation_error.to_numpy()
        mons.append(dict(seed=key[0], n=key[1], geometry=key[2], mode=key[3], steps=len(e)-1,
                         E_monotone=bool(np.all(np.diff(e)<=1e-10)), D_monotone=bool(np.all(np.diff(d)<=1e-10)),
                         orientation_monotone=bool(np.all(np.diff(o)<=1e-10)), E_increase_count=int(np.sum(np.diff(e)>1e-10))))

    source_origin = ep.groupby(['source_family','true_class','mode']).agg(cases=('E_OSC','size'), endpoint_basin_rate=('endpoint_E_basin','mean'), start_basin_rate=('start_E_basin','mean'), mean_E_OSC=('E_OSC','mean'), max_E_OSC=('E_OSC','max'), mean_E_reduction=('E_reduction_frac','mean'), mean_accepted_steps=('accepted_steps','mean'), E_monotone_rate=('E_monotone','mean')).reset_index()
    folds=[]

            folds.append(dict(axis=axis,heldout_value=str(val),rows=int(len(g)),accuracy=float(g.correct.mean()),present_class_count=int(len(present)),macro_balanced_accuracy=float(present.balanced_accuracy.mean()) if len(present) else 0.0))

    ]).to_csv(OUT/'v906_rule_comparison.csv',index=False)

    # plots

    plt.figure(figsize=(10,5))

    plt.ylabel('endpoint E_OSC'); plt.title('Endpoint energy by seven-class taxonomy'); plt.xticks(rotation=25,ha='right'); plt.tight_layout(); plt.savefig(OUT/'v906_endpoint_energy_by_class.png',dpi=170); plt.close()
    plt.figure(figsize=(8,5))
    for fam,g in ep.groupby('source_family'):
        plt.scatter(g.accepted_steps,g.E_reduction_frac,s=18,alpha=.65,label=fam)
    plt.xlabel('accepted ordered updates'); plt.ylabel('E_OSC reduction fraction'); plt.title('V906 source-family path signature space'); plt.legend(fontsize=8); plt.tight_layout(); plt.savefig(OUT/'v906_source_family_signature_space.png',dpi=170); plt.close()

    certified = bool(overall_acc==1.0 and min_ba==1.0 and seven_binary['FP']==0 and seven_binary['FN']==0 and len(false)==0)
    result=dict(
